# Remote Sensing Detection Demo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a reproducible DOTA-based three-class OBB detection baseline that trains on a rented RTX 4090, performs tiled inference on 10,000×10,000 imagery, evaluates with the competition Recall/FDR rules, exports COCO JSON, and exposes the same pipeline through CLI and Gradio.

**Architecture:** A small `xh_detect` Python package owns the shared domain types, data adapter, tiling, model adapter, cross-tile merge, evaluator, exporter, visualization, and orchestration. CLI and Gradio remain thin adapters over the same `InferencePipeline`; pure geometry and evaluation code stays independent of CUDA so it can be tested locally before renting a GPU.

**Tech Stack:** Python 3.11, uv 0.11.22, Ultralytics 8.4.71 with YOLO26-OBB, PyTorch/CUDA, NumPy, OpenCV, Shapely, PyYAML, Typer, Rich, Gradio 6.19.0, pytest, Ruff.

```python
# src/xh_detect/config.py
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

import yaml


@dataclass(frozen=True)
class PipelineConfig:
    model_path: str = "yolo26s-obb.pt"
    device: str = "0"
    image_size: int = 1024
    tile_size: int = 1024
    overlap: float = 0.2
    batch_size: int = 8
    merge_iou: float = 0.3
    edge_margin: int = 16
    half: bool = True
    class_thresholds: dict[int, float] = field(
        default_factory=lambda: {0: 0.25, 1: 0.25, 2: 0.25}
    )

    def __post_init__(self) -> None:
        if self.tile_size <= 0 or self.image_size <= 0:
            raise ValueError("image_size and tile_size must be positive")
        if not 0 <= self.overlap < 1:
            raise ValueError("overlap must be in [0, 1)")
        if not 0 <= self.merge_iou <= 1:
            raise ValueError("merge_iou must be in [0, 1]")
        if self.batch_size <= 0:
            raise ValueError("batch_size must be positive")
        if set(self.class_thresholds) != {0, 1, 2}:
            raise ValueError("class_thresholds must define class IDs 0, 1, and 2")

    @classmethod
    def from_yaml(cls, path: Path) -> "PipelineConfig":
        raw: dict[str, Any] = yaml.safe_load(path.read_text(encoding="utf-8"))
        raw["class_thresholds"] = {
            int(key): float(value) for key, value in raw["class_thresholds"].items()
        }
        return cls(**raw)
```

```python
# src/xh_detect/geometry.py
from shapely.geometry import Polygon

from xh_detect.types import Polygon4

HBB = tuple[float, float, float, float]


def obb_to_hbb(polygon: Polygon4) -> HBB:
    xs = [point[0] for point in polygon]
    ys = [point[1] for point in polygon]
    return min(xs), min(ys), max(xs), max(ys)


def clip_polygon(polygon: Polygon4, width: int, height: int) -> Polygon4:
    max_x = float(width - 1)
    max_y = float(height - 1)
    return tuple(
        (min(max(x, 0.0), max_x), min(max(y, 0.0), max_y)) for x, y in polygon
    )  # type: ignore[return-value]


def hbb_iou(left: HBB, right: HBB) -> float:
    ix1 = max(left[0], right[0])
    iy1 = max(left[1], right[1])
    ix2 = min(left[2], right[2])
    iy2 = min(left[3], right[3])
    intersection = max(0.0, ix2 - ix1) * max(0.0, iy2 - iy1)
    left_area = max(0.0, left[2] - left[0]) * max(0.0, left[3] - left[1])
    right_area = max(0.0, right[2] - right[0]) * max(0.0, right[3] - right[1])
    union = left_area + right_area - intersection
    return intersection / union if union > 0 else 0.0


def polygon_iou(left: Polygon4, right: Polygon4) -> float:
    left_shape = Polygon(left)
    right_shape = Polygon(right)
    if not left_shape.is_valid or not right_shape.is_valid:
        return 0.0
    union = left_shape.union(right_shape).area
    return left_shape.intersection(right_shape).area / union if union > 0 else 0.0
```

```python
# src/xh_detect/data/dota.py
from dataclasses import dataclass
from pathlib import Path
import os
import shutil

from PIL import Image
import yaml
from shapely.geometry import Polygon

from xh_detect.types import ObjectAnnotation, Polygon4

CLASS_MAP = {
    "plane": 0,
    "ship": 1,
    "small-vehicle": 2,
    "large-vehicle": 2,
    "small vehicle": 2,
    "large vehicle": 2,
}


@dataclass(frozen=True)
class ConversionStats:
    images: int
    targets: dict[int, int]
    invalid_lines: int
    skipped_images: int


def _parse_label_file(path: Path, image_id: str) -> tuple[list[ObjectAnnotation], int]:
    annotations: list[ObjectAnnotation] = []
    invalid_lines = 0
    for line in path.read_text(encoding="utf-8-sig").splitlines():
        parts = line.split()
        if len(parts) < 10 or parts[0].endswith(":GoogleEarth") or parts[0].startswith("gsd"):
            continue
        class_name = " ".join(parts[8:-1]).lower()
        if class_name not in CLASS_MAP:
            continue
        try:
            coordinates = [float(value) for value in parts[:8]]
        except ValueError:
            invalid_lines += 1
            continue
        polygon: Polygon4 = tuple(
            (coordinates[index], coordinates[index + 1]) for index in range(0, 8, 2)
        )  # type: ignore[assignment]
        annotations.append(
            ObjectAnnotation(
                image_id=image_id,
                class_id=CLASS_MAP[class_name],
                polygon=polygon,
                difficult=parts[-1] == "1",
            )
        )
    return annotations, invalid_lines


def parse_label_file(path: Path, image_id: str) -> list[ObjectAnnotation]:
    return _parse_label_file(path, image_id)[0]


def _link_or_copy(source: Path, destination: Path) -> None:
    destination.parent.mkdir(parents=True, exist_ok=True)
    if destination.exists() or destination.is_symlink():
        destination.unlink()
    try:
        os.link(source, destination)
    except OSError:
        try:
            destination.symlink_to(source.resolve())
        except OSError:
            shutil.copy2(source, destination)


def _to_yolo_line(annotation: ObjectAnnotation, width: int, height: int) -> str:
    values: list[str] = [str(annotation.class_id)]
    for x, y in annotation.polygon:
        values.extend((f"{x / width:.8f}", f"{y / height:.8f}"))
    return " ".join(values)


def _valid_annotation(annotation: ObjectAnnotation, width: int, height: int) -> bool:
    if Polygon(annotation.polygon).area <= 0:
        return False
    return all(0 <= x < width and 0 <= y < height for x, y in annotation.polygon)


def convert_split(
    images_dir: Path,
    labels_dir: Path,
    output_root: Path,
    split: str,
) -> ConversionStats:
    targets = {0: 0, 1: 0, 2: 0}
    invalid_lines = 0
    skipped_images = 0
    converted_images = 0
    image_paths = sorted(images_dir.glob("*.png"))
    for image_path in image_paths:
        try:
            with Image.open(image_path) as image:
                width, height = image.size
                image.verify()
        except OSError:
            skipped_images += 1
            continue
        destination_image = output_root / "images" / split / image_path.name
        _link_or_copy(image_path, destination_image)
        label_path = labels_dir / f"{image_path.stem}.txt"
        if label_path.exists():
            annotations, parse_errors = _parse_label_file(label_path, image_path.stem)
            invalid_lines += parse_errors
        else:
            annotations = []
        valid = [item for item in annotations if _valid_annotation(item, width, height)]
        invalid_lines += len(annotations) - len(valid)
        usable = [item for item in valid if not item.difficult]
        for item in usable:
            targets[item.class_id] += 1
        destination_label = output_root / "labels" / split / f"{image_path.stem}.txt"
        destination_label.parent.mkdir(parents=True, exist_ok=True)
        text = "\n".join(_to_yolo_line(item, width, height) for item in usable)
        destination_label.write_text(f"{text}\n" if text else "", encoding="utf-8")
        converted_images += 1
    return ConversionStats(converted_images, targets, invalid_lines, skipped_images)


def write_dataset_yaml(output_root: Path) -> Path:
    dataset_yaml = output_root / "dataset.yaml"
    payload = {
        "path": str(output_root.resolve()),
        "train": "images/train",
        "val": "images/val",
        "names": {0: "aircraft", 1: "ship", 2: "vehicle"},
    }
    dataset_yaml.write_text(yaml.safe_dump(payload, sort_keys=False), encoding="utf-8")
    return dataset_yaml
```

```python
# src/xh_detect/tiling.py
from collections.abc import Iterator

import numpy as np

from xh_detect.types import ImageArray, Tile, TileMeta


def axis_positions(length: int, tile_size: int, stride: int) -> list[int]:
    if length <= tile_size:
        return [0]
    positions = list(range(0, length - tile_size + 1, stride))
    final = length - tile_size
    if positions[-1] != final:
        positions.append(final)
    return positions


def iter_tiles(
    image: ImageArray,
    image_id: str,
    tile_size: int,
    overlap: float,
    pad_value: int = 114,
) -> Iterator[Tile]:
    height, width = image.shape[:2]
    stride = max(1, round(tile_size * (1 - overlap)))
    for y in axis_positions(height, tile_size, stride):
        for x in axis_positions(width, tile_size, stride):
            crop = image[y : y + tile_size, x : x + tile_size]
            valid_height, valid_width = crop.shape[:2]
            tile_image = np.full((tile_size, tile_size, 3), pad_value, dtype=np.uint8)
            tile_image[:valid_height, :valid_width] = crop
            tile_id = f"{image_id}__x{x}_y{y}_s{tile_size}"
            yield Tile(
                image=tile_image,
                meta=TileMeta(
                    image_id=image_id,
                    tile_id=tile_id,
                    x=x,
                    y=y,
                    width=tile_size,
                    height=tile_size,
                    valid_width=valid_width,
                    valid_height=valid_height,
                ),
            )
```

```python
# src/xh_detect/detector.py
from collections.abc import Sequence
from typing import Protocol

import torch
from ultralytics import YOLO

from xh_detect.types import BoxPrediction, ImageArray, Polygon4


class Detector(Protocol):
    def predict(
        self, images: list[ImageArray], confidence: float
    ) -> list[list[BoxPrediction]]:
        raise NotImplementedError


class UltralyticsOBBDetector:
    def __init__(
        self,
        model_path: str,
        device: str,
        image_size: int,
        half: bool,
    ) -> None:
        self.model = YOLO(model_path)
        self.device = device
        self.image_size = image_size
        self.half = half

    def predict(
        self, images: list[ImageArray], confidence: float
    ) -> list[list[BoxPrediction]]:
        results = self.model.predict(
            source=images,
            imgsz=self.image_size,
            conf=confidence,
            device=self.device,
            half=self.half,
            verbose=False,
        )
        batches: list[list[BoxPrediction]] = []
        for result in results:
            predictions: list[BoxPrediction] = []
            if result.obb is not None:
                polygons = result.obb.xyxyxyxy.detach().cpu().numpy()
                classes = result.obb.cls.detach().cpu().numpy().astype(int)
                scores = result.obb.conf.detach().cpu().numpy()
                for polygon, class_id, score in zip(polygons, classes, scores, strict=True):
                    points: Polygon4 = tuple(
                        (float(point[0]), float(point[1])) for point in polygon
                    )  # type: ignore[assignment]
                    predictions.append(BoxPrediction(int(class_id), float(score), points))
            batches.append(predictions)
        return batches


def predict_with_oom_backoff(
    detector: Detector,
    images: Sequence[ImageArray],
    confidence: float,
    initial_batch_size: int,
) -> list[list[BoxPrediction]]:
    output: list[list[BoxPrediction]] = []
    index = 0
    batch_size = min(initial_batch_size, len(images)) if images else 1
    while index < len(images):
        chunk = list(images[index : index + batch_size])
        try:
            output.extend(detector.predict(chunk, confidence))
            index += len(chunk)
        except RuntimeError as error:
            if "out of memory" not in str(error).lower() or batch_size == 1:
                raise
            batch_size = max(1, batch_size // 2)
            if torch.cuda.is_available():
                torch.cuda.empty_cache()
    return output
```

```python
# src/xh_detect/merge.py
from collections import defaultdict
from collections.abc import Iterable

from xh_detect.geometry import clip_polygon, hbb_iou, obb_to_hbb, polygon_iou
from xh_detect.types import BoxPrediction, Detection, TileMeta


def project_prediction(
    prediction: BoxPrediction,
    meta: TileMeta,
    image_width: int,
    image_height: int,
) -> Detection:
    shifted = tuple(
        (x + meta.x, y + meta.y) for x, y in prediction.polygon
    )
    return Detection(
        image_id=meta.image_id,
        class_id=prediction.class_id,
        score=prediction.score,
        polygon=clip_polygon(shifted, image_width, image_height),  # type: ignore[arg-type]
    )


def keep_tile_prediction(
    prediction: BoxPrediction,
    meta: TileMeta,
    image_width: int,
    image_height: int,
    margin: int,
) -> bool:
    xmin, ymin, xmax, ymax = obb_to_hbb(prediction.polygon)
    center_x = (xmin + xmax) / 2
    center_y = (ymin + ymax) / 2
    if not (0 <= center_x < meta.valid_width and 0 <= center_y < meta.valid_height):
        return False
    touches_left_internal = meta.x > 0 and xmin < margin
    touches_top_internal = meta.y > 0 and ymin < margin
    touches_right_internal = (
        meta.x + meta.valid_width < image_width and xmax > meta.valid_width - margin
    )
    touches_bottom_internal = (
        meta.y + meta.valid_height < image_height and ymax > meta.valid_height - margin
    )
    return not (
        touches_left_internal
        or touches_top_internal
        or touches_right_internal
        or touches_bottom_internal
    )


def merge_detections(
    detections: Iterable[Detection],
    iou_threshold: float,
) -> list[Detection]:
    grouped: dict[tuple[str, int], list[Detection]] = defaultdict(list)
    for detection in detections:
        grouped[(detection.image_id, detection.class_id)].append(detection)

    kept: list[Detection] = []
    for group in grouped.values():
        remaining = sorted(group, key=lambda item: item.score, reverse=True)
        while remaining:
            selected = remaining.pop(0)
            kept.append(selected)
            selected_hbb = obb_to_hbb(selected.polygon)
            survivors: list[Detection] = []
            for candidate in remaining:
                candidate_hbb = obb_to_hbb(candidate.polygon)
                if hbb_iou(selected_hbb, candidate_hbb) == 0:
                    survivors.append(candidate)
                    continue
                if polygon_iou(selected.polygon, candidate.polygon) < iou_threshold:
                    survivors.append(candidate)
            remaining = survivors
    return sorted(kept, key=lambda item: item.score, reverse=True)
```

```python
from xh_detect.types import BoxPrediction, Polygon4


class TilePredictionCache:
    def __init__(self, root: Path) -> None:
        self.root = root
        self.root.mkdir(parents=True, exist_ok=True)

    def _path(self, tile_id: str) -> Path:
        return self.root / f"{tile_id}.json"

    def load(self, tile_id: str) -> list[BoxPrediction] | None:
        path = self._path(tile_id)
        if not path.exists():
            return None
        payload = json.loads(path.read_text(encoding="utf-8"))
        return [
            BoxPrediction(
                class_id=int(item["class_id"]),
                score=float(item["score"]),
                polygon=tuple(tuple(point) for point in item["polygon"]),  # type: ignore[arg-type]
            )
            for item in payload
        ]

    def save(self, tile_id: str, predictions: list[BoxPrediction]) -> None:
        payload = [
            {
                "class_id": item.class_id,
                "score": item.score,
                "polygon": item.polygon,
            }
            for item in predictions
        ]
        destination = self._path(tile_id)
        temporary = destination.with_suffix(".tmp")
        temporary.write_text(json.dumps(payload), encoding="utf-8")
        temporary.replace(destination)
```

```python
# src/xh_detect/pipeline.py
from dataclasses import asdict
import hashlib
import json
from pathlib import Path
from time import perf_counter

from xh_detect.cache import TilePredictionCache
from xh_detect.config import PipelineConfig
from xh_detect.detector import Detector, predict_with_oom_backoff
from xh_detect.merge import keep_tile_prediction, merge_detections, project_prediction
from xh_detect.tiling import iter_tiles
from xh_detect.types import Detection, ImageArray, InferenceResult, StageTimings, Tile


class InferencePipeline:
    def __init__(
        self,
        detector: Detector,
        config: PipelineConfig,
        cache_root: Path | None = None,
    ) -> None:
        self.detector = detector
        self.config = config
        self.cache = (
            TilePredictionCache(cache_root / _cache_namespace(config))
            if cache_root
            else None
        )

    def run(self, image: ImageArray, image_id: str) -> InferenceResult:
        total_start = perf_counter()
        preprocess_start = perf_counter()
        tiles = list(
            iter_tiles(
                image,
                image_id=image_id,
                tile_size=self.config.tile_size,
                overlap=self.config.overlap,
            )
        )
        preprocess_s = perf_counter() - preprocess_start

        inference_start = perf_counter()
        predictions_by_tile: dict[str, list] = {}
        pending: list[Tile] = []
        for tile in tiles:
            cached = self.cache.load(tile.meta.tile_id) if self.cache else None
            if cached is None:
                pending.append(tile)
            else:
                predictions_by_tile[tile.meta.tile_id] = cached
        for start in range(0, len(pending), self.config.batch_size):
            batch = pending[start : start + self.config.batch_size]
            confidence = min(self.config.class_thresholds.values())
            outputs = predict_with_oom_backoff(
                self.detector,
                [tile.image for tile in batch],
                confidence=confidence,
                initial_batch_size=self.config.batch_size,
            )
            for tile, predictions in zip(batch, outputs, strict=True):
                filtered = [
                    item
                    for item in predictions
                    if item.class_id in self.config.class_thresholds
                    and item.score >= self.config.class_thresholds[item.class_id]
                ]
                predictions_by_tile[tile.meta.tile_id] = filtered
                if self.cache:
                    self.cache.save(tile.meta.tile_id, filtered)
        inference_s = perf_counter() - inference_start

        postprocess_start = perf_counter()
        height, width = image.shape[:2]
        projected: list[Detection] = []
        for tile in tiles:
            for prediction in predictions_by_tile[tile.meta.tile_id]:
                if keep_tile_prediction(
                    prediction,
                    tile.meta,
                    image_width=width,
                    image_height=height,
                    margin=self.config.edge_margin,
                ):
                    projected.append(project_prediction(prediction, tile.meta, width, height))
        detections = tuple(merge_detections(projected, self.config.merge_iou))
        postprocess_s = perf_counter() - postprocess_start
        total_s = perf_counter() - total_start
        return InferenceResult(
            detections=detections,
            timings=StageTimings(preprocess_s, inference_s, postprocess_s, total_s),
        )


def _cache_namespace(config: PipelineConfig) -> str:
    model_path = Path(config.model_path)
    payload = {"config": asdict(config)}
    if model_path.exists():
        stat = model_path.stat()
        payload["model_size"] = stat.st_size
        payload["model_mtime_ns"] = stat.st_mtime_ns
    encoded = json.dumps(payload, sort_keys=True).encode("utf-8")
    return hashlib.sha256(encoded).hexdigest()[:16]
```

```python
# src/xh_detect/exporters.py
import json
import math
from pathlib import Path
from collections.abc import Iterable

from xh_detect.geometry import obb_to_hbb
from xh_detect.types import Detection


def validate_coco_results(records: list[dict]) -> None:
    seen: set[tuple] = set()
    for index, record in enumerate(records):
        required = {"image_id", "category_id", "bbox", "score"}
        if set(record) != required:
            raise ValueError(f"record {index} has invalid fields")
        x, y, width, height = record["bbox"]
        values = [x, y, width, height, record["score"]]
        if not all(math.isfinite(float(value)) for value in values):
            raise ValueError(f"record {index} contains non-finite values")
        if width <= 0 or height <= 0:
            raise ValueError(f"record {index} bbox dimensions must be positive")
        if not 0 <= record["score"] <= 1:
            raise ValueError(f"record {index} score must be in [0, 1]")
        identity = (
            record["image_id"],
            record["category_id"],
            tuple(round(float(value), 4) for value in record["bbox"]),
        )
        if identity in seen:
            raise ValueError(f"record {index} is an exact duplicate")
        seen.add(identity)


def export_coco_results(
    detections: Iterable[Detection],
    image_id_map: dict[str, int],
    destination: Path,
) -> Path:
    records: list[dict] = []
    for detection in detections:
        if detection.image_id not in image_id_map:
            raise ValueError(f"unknown image_id: {detection.image_id}")
        xmin, ymin, xmax, ymax = obb_to_hbb(detection.polygon)
        records.append(
            {
                "image_id": image_id_map[detection.image_id],
                "category_id": detection.class_id,
                "bbox": [xmin, ymin, xmax - xmin, ymax - ymin],
                "score": detection.score,
            }
        )
    validate_coco_results(records)
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_text(
        json.dumps(records, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )
    return destination
```

```python
# src/xh_detect/evaluator.py
import json
from dataclasses import dataclass
from collections import defaultdict
from collections.abc import Iterable
from pathlib import Path

from xh_detect.geometry import hbb_iou, obb_to_hbb
from xh_detect.types import Detection, ObjectAnnotation, Polygon4

IOU_THRESHOLDS = {0: 0.50, 1: 0.50, 2: 0.35}


@dataclass(frozen=True)
class Metrics:
    tp: int
    fp: int
    fn: int

    @property
    def recall(self) -> float:
        denominator = self.tp + self.fn
        return self.tp / denominator if denominator else 0.0

    @property
    def fdr(self) -> float:
        denominator = self.fp + self.tp
        return self.fp / denominator if denominator else 0.0


@dataclass(frozen=True)
class EvaluationReport:
    overall: Metrics
    by_class: dict[int, Metrics]
    by_image: dict[str, Metrics]


def evaluate(
    predictions: Iterable[Detection],
    ground_truth: Iterable[ObjectAnnotation],
) -> EvaluationReport:
    truth_by_key: dict[tuple[str, int], list[ObjectAnnotation]] = defaultdict(list)
    for item in ground_truth:
        if not item.difficult:
            truth_by_key[(item.image_id, item.class_id)].append(item)

    matched: dict[tuple[str, int], set[int]] = defaultdict(set)
    counts = {class_id: [0, 0, 0] for class_id in (0, 1, 2)}
    image_counts: dict[str, list[int]] = defaultdict(lambda: [0, 0, 0])
    for prediction in sorted(predictions, key=lambda item: item.score, reverse=True):
        key = (prediction.image_id, prediction.class_id)
        candidates = truth_by_key[key]
        best_index = -1
        best_iou = 0.0
        for index, truth in enumerate(candidates):
            if index in matched[key]:
                continue
            iou = hbb_iou(obb_to_hbb(prediction.polygon), obb_to_hbb(truth.polygon))
            if iou > best_iou:
                best_iou = iou
                best_index = index
        if best_index >= 0 and best_iou >= IOU_THRESHOLDS[prediction.class_id]:
            matched[key].add(best_index)
            counts[prediction.class_id][0] += 1
            image_counts[prediction.image_id][0] += 1
        else:
            counts[prediction.class_id][1] += 1
            image_counts[prediction.image_id][1] += 1

    for (image_id, class_id), truths in truth_by_key.items():
        missed = len(truths) - len(matched[(image_id, class_id)])
        counts[class_id][2] += missed
        image_counts[image_id][2] += missed

    by_class = {
        class_id: Metrics(tp=values[0], fp=values[1], fn=values[2])
        for class_id, values in counts.items()
    }
    overall = Metrics(
        tp=sum(item.tp for item in by_class.values()),
        fp=sum(item.fp for item in by_class.values()),
        fn=sum(item.fn for item in by_class.values()),
    )
    by_image = {
        image_id: Metrics(tp=values[0], fp=values[1], fn=values[2])
        for image_id, values in image_counts.items()
    }
    return EvaluationReport(overall=overall, by_class=by_class, by_image=by_image)


def threshold_sweep(
    predictions: list[Detection],
    ground_truth: list[ObjectAnnotation],
    thresholds: list[float],
) -> list[tuple[float, EvaluationReport]]:
    return [
        (
            threshold,
            evaluate([item for item in predictions if item.score >= threshold], ground_truth),
        )
        for threshold in thresholds
    ]


def _bbox_to_polygon(bbox: list[float]) -> Polygon4:
    x, y, width, height = map(float, bbox)
    return (
        (x, y),
        (x + width, y),
        (x + width, y + height),
        (x, y + height),
    )


def load_coco_predictions(path: Path) -> list[Detection]:
    payload = json.loads(path.read_text(encoding="utf-8"))
    return [
        Detection(
            image_id=str(item["image_id"]),
            class_id=int(item["category_id"]),
            score=float(item["score"]),
            polygon=_bbox_to_polygon(item["bbox"]),
        )
        for item in payload
    ]


def load_coco_ground_truth(path: Path) -> list[ObjectAnnotation]:
    payload = json.loads(path.read_text(encoding="utf-8"))
    return [
        ObjectAnnotation(
            image_id=str(item["image_id"]),
            class_id=int(item["category_id"]),
            polygon=_bbox_to_polygon(item["bbox"]),
            difficult=bool(item.get("iscrowd", 0)),
        )
        for item in payload["annotations"]
    ]


def report_to_dict(report: EvaluationReport) -> dict:
    def metrics_dict(metrics: Metrics) -> dict[str, float | int]:
        return {
            "tp": metrics.tp,
            "fp": metrics.fp,
            "fn": metrics.fn,
            "recall": metrics.recall,
            "fdr": metrics.fdr,
        }

    return {
        "overall": metrics_dict(report.overall),
        "by_class": {
            str(class_id): metrics_dict(metrics)
            for class_id, metrics in report.by_class.items()
        },
        "by_image": {
            image_id: metrics_dict(metrics)
            for image_id, metrics in report.by_image.items()
        },
    }
```

```python
from collections import Counter
from collections.abc import Iterable

import cv2
import numpy as np

from xh_detect.geometry import obb_to_hbb
from xh_detect.types import Detection, ImageArray

CLASS_NAMES = {0: "aircraft", 1: "ship", 2: "vehicle"}
COLORS = {0: (0, 255, 255), 1: (255, 128, 0), 2: (0, 255, 0)}


def class_counts(detections: Iterable[Detection]) -> dict[str, int]:
    counts = Counter(item.class_id for item in detections)
    return {name: counts[class_id] for class_id, name in CLASS_NAMES.items()}


def draw_detections(
    image: ImageArray,
    detections: Iterable[Detection],
    mode: str = "obb",
) -> ImageArray:
    rendered = image.copy()
    for detection in detections:
        color = COLORS[detection.class_id]
        if mode == "obb":
            points = np.array(detection.polygon, dtype=np.int32).reshape((-1, 1, 2))
            cv2.polylines(rendered, [points], isClosed=True, color=color, thickness=2)
            x, y = map(int, detection.polygon[0])
        elif mode == "hbb":
            xmin, ymin, xmax, ymax = map(int, obb_to_hbb(detection.polygon))
            cv2.rectangle(rendered, (xmin, ymin), (xmax, ymax), color, 2)
            x, y = xmin, ymin
        else:
            raise ValueError("mode must be 'obb' or 'hbb'")
        label = f"{CLASS_NAMES[detection.class_id]} {detection.score:.2f}"
        cv2.putText(rendered, label, (x, max(15, y)), cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 1)
    return rendered
```

```python
# src/xh_detect/benchmark.py
from pathlib import Path
from statistics import median

import cv2
import numpy as np

from xh_detect.pipeline import InferencePipeline


def summarize_durations(values: list[float]) -> dict[str, float]:
    if not values:
        raise ValueError("at least one duration is required")
    return {
        "median_s": float(median(values)),
        "p95_s": float(np.percentile(values, 95)),
    }


def create_synthetic_image(
    destination: Path,
    width: int = 10_000,
    height: int = 10_000,
) -> Path:
    image = np.full((height, width, 3), 96, dtype=np.uint8)
    for x in range(500, width, 1500):
        cv2.rectangle(image, (x, 500), (min(x + 300, width - 1), 700), (140, 140, 140), -1)
    destination.parent.mkdir(parents=True, exist_ok=True)
    cv2.imwrite(str(destination), image)
    return destination


def benchmark_pipeline(
    pipeline: InferencePipeline,
    image: np.ndarray,
    image_id: str,
    repeats: int,
) -> dict[str, float]:
    pipeline.run(image, f"{image_id}-warmup")
    durations = [
        pipeline.run(image, f"{image_id}-{index}").timings.total_s
        for index in range(repeats)
    ]
    return summarize_durations(durations)
```

```python
    image_path: Path = typer.Option(Path("outputs/benchmark/synthetic-10000.png")),
    repeats: int = 5,
) -> None:
    if not image_path.exists():
        create_synthetic_image(image_path)
    image = cv2.imread(str(image_path))

    config = PipelineConfig.from_yaml(config_path)

    )
    pipeline = InferencePipeline(detector, config, cache_root=None)
    typer.echo(json.dumps(benchmark_pipeline(pipeline, image, image_path.stem, repeats)))
```
